cmake_minimum_required(VERSION 3.21)

project(DeadVault
    VERSION 0.1.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

enable_testing()

add_library(vault_core
    src/core/vault.cpp
)

target_include_directories(vault_core
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

add_executable(deadvault
    src/app/main.cpp
)

target_link_libraries(deadvault
    PRIVATE vault_core
)

add_executable(vault_tests
    tests/vault_tests.cpp
)

target_link_libraries(vault_tests
    PRIVATE vault_core
)

add_test(
    NAME vault_tests
    COMMAND vault_tests
)
